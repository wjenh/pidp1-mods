# Source-tree refactor tracking file (src/blincolnlights, phase 1 + phase 2)
# Started 16-Jul-2026.
#
# Purpose: record every file/directory operation from this refactor pass as a flat list,
# so the same operations can later be replayed (by hand, or via a generated script) against
# a parallel copy of this repository. Format is one operation per line:
#   DELETE <path>              remove this file or (now-empty) directory
#   COPY <old-path> <new-path> copy old-path to new-path, creating new-path's directory
#                               if needed (content already verified identical at time
#                               of writing; new-path may have had small path-only edits,
#                               noted below)
#
# STATUS NOTE: this Cowork sandbox session was not able to actually delete files inside
# the connected pidp1-mods folder -- rm fails with "Operation not permitted", and the
# delete-permission tool that normally grants this only recognizes the scratch outputs
# folder, not connected project folders. So the COPY lines below (the Archive relocation)
# have already been done. The DELETE lines have NOT been done yet and need to be run by
# hand (or by whoever executes this file as a script) from the repository root:
#
#   xargs -I{} rm -v {} < <(grep '^DELETE ' REFACTOR-CHANGES.txt | cut -d' ' -f2)
#
# or just run the individual rm/rmdir commands listed under each DELETE line's comment.

# --- Phase 1: dead code removal ---
# Each of these was confirmed via grep (source, headers, all Makefiles) to have zero
# references anywhere else in src/blincolnlights before being listed here.

# gpio.c/gpio.h: direct BCM2835-only register access, superseded by the Pinctrl library
# (panel_pidp1/Pinctrl), which also supports BCM2711/2712/RP1. Nothing includes gpio.h.
DELETE src/blincolnlights/gpio.c
DELETE src/blincolnlights/gpio.h

# checkConfig.c: old ad hoc config-file reader (getConfig/checkConfig), fully superseded
# by pdp1/configuration.c. Not referenced by any Makefile or source file.
DELETE src/blincolnlights/checkConfig.c

# pinctrl.c: vendored copy of the standalone "pinctrl" CLI tool. Entry point is xmain(),
# not main(); this directory's Makefile only builds .o files with no link step, so this
# file produces no executable. Only gpiolib.c/gpiochip_*.c/util.c from this directory are
# actually linked into newpanel.
DELETE src/blincolnlights/panel_pidp1/Pinctrl/pinctrl.c

# mkptyfio_telnet.c: no Makefile target anywhere in src/blincolnlights (tools/Makefile only
# builds mkptyfio and mkptyfl). NOTE: install/install.sh (outside this refactor's scope)
# still symlinks a binary by this name -- that reference will dangle after this delete and
# was already stale before it (the file was never built by this Makefile in the first
# place). Flagging again since it contradicts "no known use" at the install-script level;
# the project owner reviewed this and asked for the delete anyway.
DELETE src/blincolnlights/tools/mkptyfio_telnet.c

# --- Phase 2: panel_pidp1/Old relocation ---
# Old/ was a previous-generation panel_pidp1.c, not invoked by any parent Makefile, kept
# only as a historical comparison point for the panel-flicker investigation referenced in
# panel_pidp1/CLAUDE.md. Moved out of the live build tree into Archive/.
#
# Content is byte-identical except: Makefile's INC/PINCTRL paths were updated for the new
# directory depth (../Pinctrl -> ../../panel_pidp1/Pinctrl, since the sibling directory is
# no longer panel_pidp1), and setpriv.sh's two hardcoded paths were updated to the new
# location. panel_pidp1.c itself is unchanged except for one added comment noting the move.
COPY src/blincolnlights/panel_pidp1/Old/panel_pidp1.c src/blincolnlights/Archive/panel_pidp1_Old/panel_pidp1.c
COPY src/blincolnlights/panel_pidp1/Old/Makefile src/blincolnlights/Archive/panel_pidp1_Old/Makefile
COPY src/blincolnlights/panel_pidp1/Old/setpriv.sh src/blincolnlights/Archive/panel_pidp1_Old/setpriv.sh

# Old copies at the original location, plus the now-redundant directory, still need removal:
DELETE src/blincolnlights/panel_pidp1/Old/panel_pidp1.c
DELETE src/blincolnlights/panel_pidp1/Old/Makefile
DELETE src/blincolnlights/panel_pidp1/Old/setpriv.sh
DELETE src/blincolnlights/panel_pidp1/Old
